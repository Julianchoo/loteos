```typescript
// Generate SEO-friendly slugs
function generateProjectSlug(projectName: string, location: string): string {
  const normalize = (str: string) =>
    str.toLowerCase()
       .normalize('NFD').replace(/[\u0300-\u036f]/g, '') // remove accents
       .replace(/[^a-z0-9]+/g, '-') // replace spaces/special chars with hyphens
       .replace(/^-+|-+$/g, ''); // remove leading/trailing hyphens

  return `/proyectos/${normalize(projectName)}-${normalize(location)}`;
}

// Example:
generateProjectSlug("Lomas del Río Luján", "Campana")
// Returns: /proyectos/lomas-del-rio-lujan-campana
```
